\band{accent2}{2\quad THE WOW — the LLM is in the loop \emph{twice}, and stopping is \emph{measured}}

\begin{multicols}{2}
{\footnotesize
\textbf{Seed.} One LLM call $\rightarrow$ 5 diverse queries (6 in opinion mode: half support, half challenge).\\[2pt]
\textbf{Fetch.} All pending \code{(query,source)} in a thread pool; per-query limit shrinks so a run never blows the 500-post budget.\\[2pt]
\textbf{Dedup.} 64-bit SimHash, Hamming $\le10$ — reposts don't burn tokens or double-count clusters.\\[2pt]
\textbf{Gate.} Drop posts with topic-overlap $<0.12$ — \emph{but only if $\ge6$ survive} (recall floor).\\[2pt]
\textbf{Cluster.} Gemini embeds $\rightarrow$ HDBSCAN (finds $k$ itself); KMeans fallback.\\[2pt]
\textbf{Decide.} Labels (not raw posts) $\rightarrow$ \code{\_llm\_next} proposes $\le3$ queries, each re-anchored to the core subject so it can't drift.
}

\band{warn}{4\quad WHY NOTHING SINGLE-POINTS-OF-FAILURE \& THE NUMBERS}

\begin{multicols}{2}
{\footnotesize
\textbf{Resilience chain (every layer degrades, none crashes):}
\begin{itemize}
\item connector fails $\rightarrow$ log \& skip
\item HDBSCAN can't cluster $\rightarrow$ KMeans
\item Gemini key 429 / spend-cap $\rightarrow$ backup key, sticks
\item pipeline throws $\rightarrow$ graph routes to \code{recover} (retry$\times$3, 30s backoff)
\end{itemize}
\textbf{9 connectors} behind one ABC (reddit, hn, fb, x, ig, youtube, polymarket, github, bluesky). Reddit+HN always-on; FB needs cookies; X/IG return \code{[]} until keyed.}

\band{ink}{5\quad 30-SECOND CLOSE — the five lines}

{\footnotesize
\textbf{1. Two layers.} LangGraph = resilience; \code{run\_agent} = intelligence. Don't conflate.\quad
\textbf{2. LLM in the loop twice.} Seeds queries, then re-plans from the clusters it just found — that's agentic, not a scrape.\quad
\textbf{3. Stopping is measured.} Entropy + saturation decide convergence.\quad
\textbf{4. Ranking is relevance-dominant by evidence.} +38\% over the best OSS engine, same conditions.\quad
\textbf{5. Nothing single-points.} Connector $\to$ skip; HDBSCAN $\to$ KMeans; key $\to$ backup; throw $\to$ recover.}
